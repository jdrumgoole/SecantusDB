"""One client selecting by primary key: which threads wake, and where they run."""
import os, subprocess, sys, time, socket, tempfile, shutil, threading, collections
import psycopg

B = "/root/secantusd-pg"
SECS = float(os.environ.get("SECS", "8"))
Q = "select v from exp_t where k = %s"


def tasks(pid):
    out = {}
    try:
        tids = os.listdir(f"/proc/{pid}/task")
    except FileNotFoundError:
        return out
    for tid in tids:
        base = f"/proc/{pid}/task/{tid}"
        try:
            comm = open(base + "/comm").read().strip()
            st = dict(l.split(":", 1) for l in open(base + "/status") if ":" in l)
            sched = dict(
                (l.split(":")[0].strip(), l.split(":")[1].strip())
                for l in open(base + "/sched")
                if ":" in l
            )
            stat = open(base + "/stat").read().rsplit(")", 1)[1].split()
        except (FileNotFoundError, ProcessLookupError):
            continue
        out[int(tid)] = dict(
            comm=comm,
            vol=int(st["voluntary_ctxt_switches"]),
            invol=int(st["nonvoluntary_ctxt_switches"]),
            mig=int(sched.get("se.nr_migrations", sched.get("nr_migrations", 0))),
            cpu_ticks=int(stat[11]) + int(stat[12]),
            cpu=int(stat[36]),
        )
    return out


def snapshot(pids):
    snap = {}
    for p in pids:
        for tid, v in tasks(p).items():
            snap[tid] = v
    return snap


def run(label, dsn, server_pids_fn, client_cpus=None):
    os.sched_setaffinity(0, client_cpus or range(os.cpu_count()))
    with psycopg.connect(dsn, autocommit=True) as c, c.cursor() as cur:
        for _ in range(2000):
            cur.execute(Q, (1,)); cur.fetchone()
        pids = server_pids_fn(c)
        me = os.getpid()
        # where the busiest server thread and the client sit, sampled from another thread
        stop = threading.Event(); places = collections.Counter(); seen = collections.Counter()
        def sample():
            while not stop.is_set():
                s = snapshot(pids)
                if s:
                    hot = max(s.values(), key=lambda v: v["cpu_ticks"])
                    mine = tasks(me).get(threading.main_thread().native_id)
                    if mine:
                        places["same" if hot["cpu"] == mine["cpu"] else "apart"] += 1
                        seen[hot["cpu"]] += 1
                time.sleep(0.02)
        t = threading.Thread(target=sample, daemon=True); t.start()
        a = snapshot(pids); ca = tasks(me)[threading.main_thread().native_id]
        n = 0; end = time.monotonic() + SECS
        while time.monotonic() < end:
            cur.execute(Q, (1,)); cur.fetchone(); n += 1
        b = snapshot(pids); cb = tasks(me)[threading.main_thread().native_id]
        stop.set(); t.join()
    print(f"\n== {label}: {n / SECS:,.0f} statements/s")
    tot = sum(places.values()) or 1
    print(f"   server thread and client on one CPU in {100 * places['same'] / tot:.0f}% of samples; "
          f"server seen on {len(seen)} CPUs")
    print(f"   client          : {(cb['vol'] - ca['vol']) / n:5.2f} sleeps/stmt  "
          f"{(cb['invol'] - ca['invol']) / n:5.3f} preempt/stmt  {(cb['mig'] - ca['mig']) / n:6.4f} migrations/stmt  "
          f"cpu {(cb['cpu_ticks'] - ca['cpu_ticks']) * 1e4 / n:5.1f} us/stmt")
    rows = collections.defaultdict(lambda: [0, 0, 0, 0, 0])
    for tid, v in b.items():
        o = a.get(tid)
        if not o:
            continue
        r = rows[v["comm"]]
        r[0] += v["vol"] - o["vol"]; r[1] += v["invol"] - o["invol"]; r[2] += v["mig"] - o["mig"]
        r[3] += v["cpu_ticks"] - o["cpu_ticks"]; r[4] += 1
    for comm, r in sorted(rows.items(), key=lambda kv: -kv[1][0]):
        if r[0] + r[3] == 0:
            continue
        print(f"   {comm:16}x{r[4]:<2}: {r[0] / n:5.2f} sleeps/stmt  {r[1] / n:5.3f} preempt/stmt  "
              f"{r[2] / n:6.4f} migrations/stmt  cpu {r[3] * 1e4 / n:5.1f} us/stmt")
    return n / SECS


def prep(dsn):
    with psycopg.connect(dsn, autocommit=True) as c:
        c.execute("drop table if exists exp_t")
        c.execute("create table exp_t (k bigint primary key, v bigint)")
        c.execute("insert into exp_t values (1, 0)")


def pg_pids(c):
    main = int(open("/var/run/postgresql/16-main.pid").readline())
    kids = subprocess.run(["pgrep", "-P", str(main)], capture_output=True, text=True).stdout.split()
    return [main] + [int(k) for k in kids]


def start_secantus(prefix=()):
    store = tempfile.mkdtemp(prefix="exp-")
    s = socket.socket(); s.bind(("127.0.0.1", 0)); port = s.getsockname()[1]; s.close()
    d = subprocess.Popen([*prefix, B, store, f"127.0.0.1:{port}"],
                         stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    time.sleep(2)
    return d, store, f"host=127.0.0.1 port={port} dbname=postgres user=postgres"


if __name__ == "__main__":
    which = sys.argv[1:] or ["pg", "sd"]
    PG = "host=127.0.0.1 port=5432 dbname=postgres user=postgres"
    if "pg" in which:
        prep(PG)
        for i in range(3):
            run(f"postgres unpinned #{i}", PG, pg_pids)
    if "sd" in which:
        d, store, dsn = start_secantus()
        prep(dsn)
        for i in range(3):
            run(f"secantusd-pg unpinned #{i}", dsn, lambda c: [d.pid])
        print("PID", d.pid, dsn, flush=True)
        d.terminate(); d.wait(); shutil.rmtree(store, ignore_errors=True)
